class HitBox {
private:
    float x, y;                  //Положение объекта
    float width; float height;   //Ширина и высота

public:
    HitBox(float o_x, float o_y, float o_width, float o_height) {
		x = o_x;
		y = o_y;
		width = o_width;
		height = o_height;
    }

    float GetX() {
        return x;
    }
    float GetY() {
        return y;
    }
    float GetWidth() {
        return width;
    }
    float GetHeight() {
        return height;
    }
    Rectangle GetAll() {
        return { x, y, width, height };
    }

    void Move(int move_x, int move_y) {
		x += move_x; y += move_y;
    }
    void Max(float max_x, float max_y) {
        x = max_x; y = max_y;
    }

    //Проверяет: пересекаются ли два хитбокса хоть как-то.
    bool Intersects(float o_x, float o_y, float o_width, float o_height) {
        return
            x < o_x + o_width &&
            x + width > o_x &&
            y < o_y + o_height &&
            y + height > o_y;
    }
    //Проверяет: хитбокс полностью содержит другой хитбокс внутри себя
    bool Contains(float o_x, float o_y, float o_width, float o_height)
    {
        return
            x <= o_x &&
            x + width >= o_x + o_width &&
            y <= o_y &&
            y + height >= o_y + o_height;
    }
    bool Contains(Rectangle r)
    {
        return
            x <= r.x &&
            x + width >= r.x + r.width &&
            y <= r.y &&
            y + height >= r.y + r.height;
    }
    //Проверяет: хитбокс полностью находится внутри другого хитбокса.
    bool IsInside(float o_x, float o_y, float o_width, float o_height) {
        return
            x > o_x &&
			x + width < o_x + o_width &&
            y > o_y &&
			y + height < o_y + o_height;
    }
    //Проверяет: хитбокс полностью снаружи другого, вообще не пересекается.
    bool IsOutside(float o_x, float o_y, float o_width, float o_height) {
        return
            x + width <= o_x ||
            x >= o_x + o_width ||
            y + height <= o_y ||
            y >= o_y + o_height;
    }
    //Проверяет: хитбокс вышел за границы зоны, даже частично.
    bool IsOutOfBounds(float o_x, float o_y, float o_width, float o_height) {
        return
            x < o_x ||
			x + width > o_x + o_width ||
			y < o_y ||
			y + height > o_y + o_height;
    }

    //Функции для стен
    bool BlockMovement(float o_x, float o_y, float o_width, float o_height) {



    }
    

};

class GameObject {
protected:
	float x, y;                  //Положение объекта
    float width; float height;   //Ширина и высота
    bool active; 
    HitBox HitB;

public:
    GameObject(float o_x, float o_y, float o_width, float o_height, bool o_active = false) : HitB(o_x, o_y, o_width, o_height)  {
        x = o_x;
        y = o_y;
        width = o_width;
        height = o_height;
        active = o_active;
    }
    GameObject() : HitB(0, 0, 0, 0)
    {
        x = 0;
        y = 0;
        width = 0;
        height = 0;
        active = false;
    }

    float GetX() {
		return x;
    }
    float GetY() {
		return y;
    }
    float GetWidth() {
        return width;
    }
	float GetHeight() {
		return height;
	}
    Rectangle GetHitbox() {
        return HitB.GetAll();
    }

    bool Intersects(GameObject& m) {
        Rectangle r = m.GetHitbox();

        return HitB.Intersects(r.x, r.y, r.width, r.height);
    }
};

class Player : public GameObject {
protected:
    int speed;                   //Скорость
    Color color;                 //Свет

public:
    Player(float o_x, float o_y, float o_width, float o_height, int o_speed, Color o_color, bool o_active = false) : GameObject(o_x, o_y, o_width, o_height, o_active) 
    {
        speed = o_speed;
        color = o_color;
    }
    bool HitPlayer(World& m);

    void MoveD() {
        x += speed;
        HitB.Move(speed, 0);
    }
    void MoveA() {
        x -= speed;
        HitB.Move(-speed, 0);
    }
    void MoveW() {
        y -= speed;
        HitB.Move(0, -speed);
    }
    void MoveS() {
        y += speed;
        HitB.Move(0, speed);
    }

    void MaxD(float Width) {
        x = Width - width;
        HitB.Max(x, y);
    }
    void MaxA() {
        x = 0;
        HitB.Max(x, y);
    }
    void MaxW() {
        y = 0;
        HitB.Max(x, y);
    }
    void MaxS(float Height) {
        y = Height - height;
        HitB.Max(x, y);
    }
    void MaxWall(float w_x, float w_y) {
        if (w_x != -1) x = w_x;
		if (w_y != -1) y = w_y;
        HitB.Max(x, y);
    }

    int GetSpeed() {
        return speed;
    }
    void SpeedUp() {
        speed += 1; 
    }
    void SlowDown() {
		if (speed > 1) speed -= 1;
    }
};

class WallManager {
private:
    vector<GameObject> Walls; 
    vector<Color> color;
public:

    void newWall (float o_x, float o_y, float o_width, float o_height, bool o_active = false, Color o_color = RED) {
        Walls.emplace_back(o_x, o_y, o_width, o_height, o_active);
        color.emplace_back(o_color);
    }

    void Paint() {
        for (int i = 0; i < Walls.size(); i++) { DrawRectangle(Walls[i].GetX(), Walls[i].GetY(), Walls[i].GetWidth(), Walls[i].GetHeight(), color[i]); }
    }

    void HitWall(Player& m, bool& active, float& move, char dir) {
        move = -1; active = false; 

        for (int i = 0; i < Walls.size(); i++) {
            if (Walls[i].Intersects(m)) {
                active = true; 
                
                switch (dir) {
                case 'D': {
                    if (move == -1) move = Walls[i].GetX();
                    else { 
						if (Walls[i].GetX() < move) move = Walls[i].GetX();
                    }
                    break;
                }

				case 'A': {
                    if (move == -1) move = Walls[i].GetX() + Walls[i].GetWidth();
                    else {
                        if (Walls[i].GetX() + Walls[i].GetWidth() > move) move = Walls[i].GetX() + Walls[i].GetWidth();
                    }
                    break;
                }

				case 'W': {
					if (move == -1) move = Walls[i].GetY() + Walls[i].GetHeight();
					else {
						if (Walls[i].GetY() + Walls[i].GetHeight() > move) move = Walls[i].GetY() + Walls[i].GetHeight();
					}
                    break;
                }

				case 'S': {
					if (move == -1) move = Walls[i].GetY();
                    else {
						if (Walls[i].GetY() < move) move = Walls[i].GetY();
                    }
                    break;
                }
                }
            }
        }
    }
};

class World : public GameObject {
private:
    WallManager Walls;

    int cS, cX, cY;

    void f_GridPaint(bool xBigger) {
        if (xBigger) {
            int s = 1;

            for (; s <= cY; s++) {
                DrawRectangle(cS * s, 0, 1, cY * cS, WHITE);  // вертикальные
                DrawRectangle(0, cS * s, cX * cS, 1, WHITE);  // горизонтальные
            }

            for (; s <= cX; s++) {
                DrawRectangle(cS * s, 0, 1, cY * cS, WHITE);  // остаток
            }
        }

        if (!xBigger) {
            int s = 1;

            for (; s <= cX; s++) {
                DrawRectangle(cS * s, 0, 1, cY * cS, WHITE); // вертикальные
                DrawRectangle(0, cS * s, cX * cS, 1, WHITE); // горизонтальные
            }

            for (; s <= cY; s++) {
                DrawRectangle(0, cS * s, cX * cS, 1, WHITE); // остаток 
            }
        }
    }

public:
    World(int cellSize, int cellsX, int cellsY, bool o_active = true) : GameObject(0, 0, cellsX* cellSize, cellsY* cellSize, o_active) {
        cS = cellSize; cX = cellsX; cY = cellsY;
    }
	bool HitGridChek(Player& m) {
        return HitB.Contains(m.GetX(), m.GetY(), m.GetWidth(), m.GetHeight());
	}
    void GridPaint() {
        if (cX >= cY) { f_GridPaint(true); }
        if (cX < cY) { f_GridPaint(false); }
    }
    void GridWallPaint() {
        Walls.Paint();
    }
    void GridHitWall(Player& m, bool& active, float& move, char dir) {
        Walls.HitWall(m, active, move, dir);
    }

    void newGridWall_10(int cellsX, int cellsY, int cells_width, int cells_height, Color color) {

void GridAreaClose::BuildWall(World& W, int Thickness_grid10, Color color) {
    W.newGridWall_10(PosX, PosY,
        PosWidth + 2 * Thickness_grid10,
        Thickness_grid10,
        color
    );

    W.newGridWall_10(PosX, PosY + Thickness_grid10 + PosHeight,
        PosWidth + 2 * Thickness_grid10,
        Thickness_grid10,
        color
    );

    W.newGridWall_10(PosX, PosY + Thickness_grid10,
        Thickness_grid10,
        PosHeight,
        color
    );

    W.newGridWall_10(PosX + Thickness_grid10 + PosWidth, PosY + Thickness_grid10,
        Thickness_grid10,
        PosHeight,
        color
    );

    PosX += Thickness_grid10;
    PosY += Thickness_grid10;
}

void HitChek(Player& Pl, World& W, WallManager& S, char dir) {

    bool w_active; float w_move;

    switch (dir) {
    case 'D': {

        if (!W.HitGridChek(Pl)) {
            Pl.MaxD(W.GetWidth());
        }
        else { 
            S.HitWall(Pl, w_active, w_move, dir);
            if (w_active) {
                Pl.MaxWall(w_move - Pl.GetWidth(), -1);
            }
            W.GridHitWall(Pl, w_active, w_move, dir);
            if (w_active) {
                Pl.MaxWall(w_move - Pl.GetWidth(), -1);
            }

        }
        break;
    }
    case 'A': {
        if (!W.HitGridChek(Pl)) {
            Pl.MaxA();
        }
        else {
            S.HitWall(Pl, w_active, w_move, dir);
            if (w_active) {
                Pl.MaxWall(w_move, -1);
            }
            W.GridHitWall(Pl, w_active, w_move, dir);
            if (w_active) {
                Pl.MaxWall(w_move, -1);
            }
        }
        break;
    } 
    case 'W': {
        if (!W.HitGridChek(Pl)) {
            Pl.MaxW();
		}
        else {
            S.HitWall(Pl, w_active, w_move, dir);
            if (w_active) {
                Pl.MaxWall(-1, w_move);
            }
            W.GridHitWall(Pl, w_active, w_move, dir);
            if (w_active) {
                Pl.MaxWall(-1, w_move);
            }
        }

        break;
    }
    case 'S': {
        if (!W.HitGridChek(Pl)) {
            Pl.MaxS(W.GetHeight());
        }
        else {
            S.HitWall(Pl, w_active, w_move, dir);
            if (w_active) {
				Pl.MaxWall(-1, w_move - Pl.GetHeight());
            }
            W.GridHitWall(Pl, w_active, w_move, dir);
            if (w_active) {
                Pl.MaxWall(-1, w_move - Pl.GetHeight());
            }
        }

        break;
    }
    }
}
